cmake_minimum_required(VERSION 3.16)
project(task_rviz_smoke_renderer)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_executable(task_rviz_smoke_renderer main.cpp)
target_compile_features(task_rviz_smoke_renderer PRIVATE cxx_std_17)
ament_target_dependencies(task_rviz_smoke_renderer rviz_common rviz_rendering)
target_link_libraries(task_rviz_smoke_renderer Qt5::Widgets)
