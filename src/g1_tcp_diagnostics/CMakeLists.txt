cmake_minimum_required(VERSION 3.8)
project(g1_tcp_diagnostics)
find_package(ament_cmake REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets Test)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(std_msgs REQUIRED)
find_package(unitree_hg REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(CycloneDDS REQUIRED)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
add_library(g1_tcp_a_panel SHARED include/g1_tcp_diagnostics/tcp_a_panel.hpp src/tcp_a_panel.cpp)
target_compile_features(g1_tcp_a_panel PUBLIC cxx_std_17)
target_include_directories(g1_tcp_a_panel PUBLIC include)
ament_target_dependencies(g1_tcp_a_panel pluginlib rclcpp rviz_common std_msgs)
target_link_libraries(g1_tcp_a_panel Qt5::Widgets)
add_library(g1_camera_b_panel SHARED include/g1_tcp_diagnostics/camera_b_panel.hpp src/camera_b_panel.cpp)
target_compile_features(g1_camera_b_panel PUBLIC cxx_std_17)
target_include_directories(g1_camera_b_panel PUBLIC include)
ament_target_dependencies(g1_camera_b_panel pluginlib rclcpp rviz_common std_msgs)
target_link_libraries(g1_camera_b_panel Qt5::Widgets)
add_executable(camera_b_window src/camera_b_window.cpp)
target_link_libraries(camera_b_window g1_camera_b_panel Qt5::Widgets)
ament_target_dependencies(camera_b_window rclcpp rviz_common std_msgs)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
add_library(g1_camera_c_panel SHARED include/g1_tcp_diagnostics/camera_c_panel.hpp src/camera_c_panel.cpp)
target_compile_features(g1_camera_c_panel PUBLIC cxx_std_17)
target_include_directories(g1_camera_c_panel PUBLIC include)
ament_target_dependencies(g1_camera_c_panel pluginlib rclcpp rviz_common std_msgs)
target_link_libraries(g1_camera_c_panel Qt5::Widgets)
add_executable(camera_c_window src/camera_c_window.cpp)
target_link_libraries(camera_c_window g1_camera_c_panel Qt5::Widgets)
ament_target_dependencies(camera_c_window rclcpp rviz_common std_msgs)
install(TARGETS g1_camera_c_panel LIBRARY DESTINATION lib)
install(TARGETS camera_c_window DESTINATION lib/${PROJECT_NAME})
add_executable(camera_c_actual_reader src/camera_c_actual_reader.cpp)
target_compile_features(camera_c_actual_reader PUBLIC cxx_std_17)
ament_target_dependencies(camera_c_actual_reader rclcpp std_msgs unitree_hg)
target_link_libraries(camera_c_actual_reader nlohmann_json::nlohmann_json CycloneDDS::ddsc)
install(TARGETS camera_c_actual_reader DESTINATION lib/${PROJECT_NAME})
add_executable(tcp_a_panel_check test/panel_check.cpp)
target_link_libraries(tcp_a_panel_check g1_tcp_a_panel g1_camera_b_panel Qt5::Test)
ament_target_dependencies(tcp_a_panel_check rviz_common rclcpp std_msgs)
add_executable(tcp_a_visual_check test/rviz_visual_check.cpp)
target_link_libraries(tcp_a_visual_check Qt5::Widgets)
ament_target_dependencies(tcp_a_visual_check rviz_common rviz_rendering rclcpp)
install(TARGETS g1_tcp_a_panel LIBRARY DESTINATION lib)
install(TARGETS g1_camera_b_panel LIBRARY DESTINATION lib)
install(TARGETS camera_b_window DESTINATION lib/${PROJECT_NAME})
install(TARGETS tcp_a_panel_check DESTINATION lib/${PROJECT_NAME})
install(TARGETS tcp_a_visual_check DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY rviz launch config DESTINATION share/${PROJECT_NAME})
install(FILES plugins.xml DESTINATION share/${PROJECT_NAME})
ament_package()
